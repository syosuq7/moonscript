local Trove = { _list = {} }
function Trove:Add(o) table.insert(self._list, o); return o end
function Trove:Destroy()
    for i=#self._list,1,-1 do
        local o = self._list[i]
        pcall(function()
            if typeof(o) == "RBXScriptConnection" then o:Disconnect()
            elseif typeof(o) == "Instance" then o:Destroy()
            elseif typeof(o) == "thread" then task.cancel(o)
            elseif typeof(o) == "function" then o()
            elseif type(o) == "table" and o.Destroy then o:Destroy() end
        end)
    end
    self._list = {}
end
local trove = Trove
getgenv().GLOBAL_TROVE = trove

    pcall(function()

    end)
    local function listLuas()
        local out = {}
        local ok, list = pcall(listfiles, lua_folder)
        if ok and type(list) == "table" then
            for _, f in ipairs(list) do
                if f:sub(-4) == ".lua" or f:sub(-4) == ".txt" then
                    local name = (f:gsub("%.lua$",""):gsub("%.txt$","")):match("([^/\\]+)$")
                    if name and name ~= "" then out[#out+1] = name end
                end
            end
        end
        return out
    end
    local input = sec:CreateInput({ Name = "lua name", Flag = "lua_name", Default = "", Placeholder = "my script" })
    local dd = sec:CreateDropdown({ Name = "saved luas", Flag = "lua_list", Options = listLuas(), Default = "", AllowNull = true,

        pcall(function() dd:SetOptions(listLuas()) end)
    end
    sec:CreateButton({ Name = "save lua", Callback = function()
        local n = "script_" .. os.time()
        if writefile then pcall(writefile, lua_folder .. "/" .. n .. ".lua", "-- saved") end
        refresh(); notify("FourHub", "Saved", 2)
    end })
    sec:CreateButton({ Name = "refresh list", Callback = refresh })
end

-- Watermark
getgenv().FourHub_wm = getgenv().FourHub_wm or { ping = true, fps = true, time = true }
task.spawn(function()
    local fps = 60
    RunService.RenderStepped:Connect(function(dt)
        if dt and dt > 0 then fps = fps + (1/dt - fps) * 0.1 end
    end)
    while task.wait(0.5) do
        if getgenv().FourHub_wm_visible ~= false then
            local wm = getgenv().FourHub_wm or {}
            local parts = { "FourHub.lol" }
            if wm.fps ~= false then parts[#parts+1] = string.format("%d fps", math.floor(fps + 0.5)) end
            if wm.ping ~= false then
                local ok, v = pcall(function() return game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() end)
                if ok and v then parts[#parts+1] = string.format("%d ms", math.floor(v + 0.5)) end
            end
            if wm.time ~= false then parts[#parts+1] = os.date("%X") end
            pcall(function() Window:SetWatermark(table.concat(parts, " | ")) end)
        end
    end
end)

    if ELI.world.color_shift_top then Lighting.ColorShift_Top = ELI.world.color_shift_top_color end
    if ELI.world.color_shift_bottom then Lighting.ColorShift_Bottom = ELI.world.color_shift_bottom_color end
end))

-- Motion blur
do
    local blur = Instance.new("BlurEffect"); blur.Size = 0; blur.Enabled = false; blur.Parent = Lighting
    trove:Add(blur)
    local lastLook = Camera.CFrame.LookVector
    trove:Add(RunService.RenderStepped:Connect(function(dt)
        local c = ELI.motion_blur
        if not c.enable then blur.Enabled = false; lastLook = Camera.CFrame.LookVector; return end
        blur.Enabled = true
        local look = Camera.CFrame.LookVector
        local delta = (look - lastLook).Magnitude
        lastLook = look
        local target = math.clamp(delta / math.max(dt, 1/240) * c.sensitivity * c.intensity * 0.35, 0, 56 * c.intensity)
        blur.Size = blur.Size + (target - blur.Size) * math.min(dt * 20, 1)
    end))
end

-- Fly
trove:Add(RunService.Heartbeat:Connect(function()
    if not ELI.fly.enable then return end
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local mv = Vector3.new()
    if UserInput:IsKeyDown(Enum.KeyCode.W) then mv += Camera.CFrame.LookVector end
    if UserInput:IsKeyDown(Enum.KeyCode.S) then mv -= Camera.CFrame.LookVector end
    if UserInput:IsKeyDown(Enum.KeyCode.A) then mv -= Camera.CFrame.RightVector end
    if UserInput:IsKeyDown(Enum.KeyCode.D) then mv += Camera.CFrame.RightVector end
    if UserInput:IsKeyDown(Enum.KeyCode.Space) then mv += Vector3.new(0,1,0) end
    if UserInput:IsKeyDown(Enum.KeyCode.LeftShift) then mv -= Vector3.new(0,1,0) end
    if mv.Magnitude > 0 then
        hrp.CFrame += mv.Unit * ELI.fly.speed * RunService.Heartbeat:Wait()
    end
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
end))

-- Targeted player helpers
local deflecting = {}
local function is_deflecting(plr) return deflecting[plr] end
local function has_riot(plr) return workspace.ViewModels:FindFirstChild(plr.Name .. " - Riot Shield - Riot Shield") ~= nil end
local function check_wall(part)
    local ray = Ray.new(Camera.CFrame.Position, part.Position - Camera.CFrame.Position)
    local hit = workspace:FindPartOnRayWithIgnoreList(ray, { LocalPlayer.Character, part.Parent })
    return not hit
end

function getclosest()
    local closest = ELI.targeting.max_distance
    local player
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    for _, v in ipairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild(ELI.targeting.part) and v.Character.Humanoid.Health > 0 then
            if is_deflecting(v) then continue end
            if v:GetAttribute("TeamID") == LocalPlayer:GetAttribute("TeamID") then continue end
            if ELI.targeting.visible_only and not check_wall(v.Character[ELI.targeting.part]) then continue end
            if ELI.targeting.riot_shield and has_riot(v) then continue end
            local pos, on = Camera:WorldToViewportPoint(v.Character[ELI.targeting.part].Position)
            if on then
                local d = (center - Vector2.new(pos.X, pos.Y)).Magnitude
                if d < closest and d <= ELI.silent_aim.fov_radius then player = v; closest = d end
            end
        end
    end
    return player
end

            end
        end
    end
    return player
end

        local bot = Camera:WorldToViewportPoint(root.Position - Vector3.new(0, 1, 0))

            c.hl.OutlineColor = ELI.esp.highlight.outline_color

            c.hl.OutlineTransparency = ELI.esp.highlight.outline_transparency

        elseif c.hl then c.hl.Enabled = false end

-- Chams
do
    local hlCache = {}
    local bodyCache = {}
    local presets = {
        Neon = { material = Enum.Material.Neon },
        ForceField = { material = Enum.Material.ForceField },
        Glass = { material = Enum.Material.Glass, reflectance = 0.1, transparency = 0.2 },
        Chrome = { material = Enum.Material.Metal, reflectance = 1 },
    }
    local function drop(plr)
        if hlCache[plr] then pcall(function() hlCache[plr]:Destroy() end); hlCache[plr] = nil end
        local saved = bodyCache[plr]
        if saved then
            for part, props in pairs(saved) do
                if part.Parent then
                    part.Material = props.material; part.Color = props.color
                    part.Reflectance = props.reflectance; part.Transparency = props.transparency
                end
            end
            bodyCache[plr] = nil
        end
    end
    local function applyHighlight(plr, char)
        local hl = hlCache[plr]
        if not hl or not hl.Parent then
            hl = Instance.new("Highlight"); hl.Name = "\0"
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent = char; hlCache[plr] = hl
        end
        hl.Adornee = char
        hl.FillColor = ELI.chams.fill_color
        hl.FillTransparency = ELI.chams.fill_transparency
        hl.OutlineColor = ELI.chams.outline_color
        hl.OutlineTransparency = ELI.chams.outline_transparency
    end
    local function applyBody(plr, char)
        local preset = presets[ELI.chams.material] or presets.Neon
        local saved = bodyCache[plr] or {}; bodyCache[plr] = saved
        local tr = preset.transparency or 0
        if ELI.chams.ghost then tr = math.max(tr, 0.55) end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.Transparency < 1 then
                if not saved[part] then
                    saved[part] = { material=part.Material, color=part.Color, reflectance=part.Reflectance, transparency=part.Transparency }
                end
                part.Material = preset.material
                part.Color = ELI.chams.fill_color
                part.Reflectance = preset.reflectance or 0
                part.Transparency = tr
            end
        end
    end
    trove:Add(RunService.Heartbeat:Connect(function()
        if not ELI.chams.enable then
            for plr in pairs(hlCache) do drop(plr) end
            for plr in pairs(bodyCache) do drop(plr) end
            return
        end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local char = plr.Character
                if not char or not char:FindFirstChildOfClass("Humanoid") or char.Humanoid.Health <= 0 then drop(plr) continue end
                if ELI.chams.team_check and plr:GetAttribute("TeamID") == LocalPlayer:GetAttribute("TeamID") then drop(plr) continue end
                if ELI.chams.mode == "chams" then
                    if hlCache[plr] then drop(plr) end
                    applyBody(plr, char)
                else
                    if bodyCache[plr] then drop(plr) end
                    applyHighlight(plr, char)
                end
            end
        end
    end))
    trove:Add(function() for plr in pairs(hlCache) do drop(plr) end for plr in pairs(bodyCache) do drop(plr) end end)
end

-- Crosshair (Drawing-based)
do
    local lines, outlines = {}, {}
    for i=1,4 do
        local o = Drawing.new("Line"); o.Visible = false; o.Thickness = 3; o.Color = Color3.new(0,0,0); outlines[i] = o
        local l = Drawing.new("Line"); l.Visible = false; l.Thickness = 1; lines[i] = l
    end
    local ring = Drawing.new("Circle"); ring.Filled = false; ring.NumSides = 48; ring.Thickness = 1; ring.Visible = false
    local ringo = Drawing.new("Circle"); ringo.Filled = false; ringo.NumSides = 48; ringo.Thickness = 3; ringo.Color = Color3.new(0,0,0); ringo.Visible = false
    local dot = Drawing.new("Circle"); dot.Filled = true; dot.NumSides = 24; dot.Radius = 1; dot.Visible = false
    local clock, spin = 0, 0
    trove:Add(RunService.RenderStepped:Connect(function(dt)
        if not ELI.crosshair.enable then
            for i=1,4 do lines[i].Visible = false; outlines[i].Visible = false end
            ring.Visible = false; ringo.Visible = false; dot.Visible = false
            return
        end
        clock += dt
        if ELI.crosshair.rotating then spin = (spin + ELI.crosshair.rotating_speed * dt) % 360 end
        local rad = math.rad(ELI.crosshair.rotation + (ELI.crosshair.rotating and spin or 0))
        local cosv, sinv = math.cos(rad), math.sin(rad)
        local c = Camera.ViewportSize
        local anchor = Vector2.new(c.X/2, c.Y/2)
        local len, gap = ELI.crosshair.length, ELI.crosshair.gap
        if ELI.crosshair.animation then
            local wave = (math.sin(clock * ELI.crosshair.animation_speed) + 1) * 0.5
            if ELI.crosshair.animation_mode == "breathe" then gap = gap + wave * len * 0.6
            else len = len * (0.55 + wave * 0.45) end
        end
        local color = ELI.crosshair.color
        local arms = {
            { Vector2.new(0,-gap), Vector2.new(0,-gap-len), ELI.crosshair.top_line },
            { Vector2.new(0,gap), Vector2.new(0,gap+len), ELI.crosshair.bottom_line },
            { Vector2.new(-gap,0), Vector2.new(-gap-len,0), ELI.crosshair.left_line },
            { Vector2.new(gap,0), Vector2.new(gap+len,0), ELI.crosshair.right_line },
        }
        if ELI.crosshair.style == "dot" then
            for i=1,4 do lines[i].Visible = false; outlines[i].Visible = false end
            ring.Visible = false; ringo.Visible = false
            dot.Position = anchor; dot.Color = color; dot.Radius = math.max(ELI.crosshair.thickness, 1); dot.Visible = true
        elseif ELI.crosshair.style == "circle" then
            for i=1,4 do lines[i].Visible = false; outlines[i].Visible = false end
            dot.Visible = false
            ring.Position = anchor; ring.Radius = len; ring.Color = color; ring.Thickness = ELI.crosshair.thickness; ring.Visible = true
            if ELI.crosshair.outline then
                ringo.Position = anchor; ringo.Radius = len; ringo.Color = ELI.crosshair.outline_color; ringo.Thickness = ELI.crosshair.thickness + 2; ringo.Visible = true
            else ringo.Visible = false end
        else
            ring.Visible = false; ringo.Visible = false; dot.Visible = false
            for i=1,4 do
                local a = arms[i]
                if ELI.crosshair.lines and a[3] then
                    local f, t = a[1], a[2]
                    local fr = Vector2.new(f.X * cosv - f.Y * sinv, f.X * sinv + f.Y * cosv)
                    local tr = Vector2.new(t.X * cosv - t.Y * sinv, t.X * sinv + t.Y * cosv)
                    lines[i].From = anchor + fr; lines[i].To = anchor + tr
                    lines[i].Color = color; lines[i].Thickness = ELI.crosshair.thickness; lines[i].Visible = true
                    if ELI.crosshair.outline then
                        outlines[i].From = anchor + fr; outlines[i].To = anchor + tr
                        outlines[i].Color = ELI.crosshair.outline_color
                        outlines[i].Thickness = ELI.crosshair.thickness + 2; outlines[i].Visible = true
                    else outlines[i].Visible = false end
                else lines[i].Visible = false; outlines[i].Visible = false end
            end
        end
    end))
    trove:Add(function()
        for i=1,4 do pcall(function() lines[i]:Remove() end); pcall(function() outlines[i]:Remove() end) end
        pcall(function() ring:Remove() end); pcall(function() ringo:Remove() end); pcall(function() dot:Remove() end)
    end)
end

        local animator = hum and hum:FindFirstChildOfClass("Animator")
        if not animator then stop() return end
        if animobj and not animobj.IsPlaying then animobj = nil; lastid = "" end
        local id = anims[ELI.animation.select_animation]
        if not id then return end
        local speed = ELI.animation.speed
        if id ~= lastid then
            if animobj then pcall(function() animobj:Stop() end); animobj = nil end
            lastid = id
            local a = Instance.new("Animation"); a.AnimationId = "rbxassetid://" .. id
            local track = animator:LoadAnimation(a)
            animobj = track
            pcall(function() track.Priority = Enum.AnimationPriority.Action4; track:Play(); track:AdjustSpeed(speed) end)
            lastspeed = speed
        end
        if animobj and speed ~= lastspeed then
            pcall(function() animobj:AdjustSpeed(speed) end)
            lastspeed = speed
        end
    end))
end
